        target_include_directories(segacd_bios_dual_probe PRIVATE "${SEGACD_BIOS_PROBE_SUB_PATCH_MODULE_DIR}")
        target_compile_definitions(segacd_bios_dual_probe PRIVATE SEGACD_BIOS_PROBE_SUB_PATCH=1)
      endif()
      # Original disc-loaded bootstrap catalogs are private generated inputs.
      # They describe source bytes; the running BIOS must load those bytes.
      foreach(cpu IN ITEMS MAIN SUB)
        set(SEGACD_BIOS_PROBE_BOOT_${cpu}_MODULE_DIR "" CACHE PATH
          "Private original disc boot ${cpu} module directory")
        if(SEGACD_BIOS_PROBE_BOOT_${cpu}_MODULE_DIR)
          string(TOLOWER "${cpu}" cpu_lower)
          set(boot_path "${SEGACD_BIOS_PROBE_BOOT_${cpu}_MODULE_DIR}")
          set(boot_prefix "segacd_boot_${cpu_lower}")
          set(boot_sources "${boot_path}/${boot_prefix}_module.c")
          foreach(part RANGE 0 31)
            if(part LESS 10)
              set(part "0${part}")
            endif()
            list(APPEND boot_sources "${boot_path}/${boot_prefix}_part${part}.c")
          endforeach()
          target_sources(segacd_bios_dual_probe PRIVATE ${boot_sources})
          target_include_directories(segacd_bios_dual_probe PRIVATE "${boot_path}")
          target_compile_definitions(segacd_bios_dual_probe PRIVATE
            SEGACD_BIOS_PROBE_BOOT_${cpu}=1)
        endif()
      endforeach()

      if(SEGACD_BUILD_NATIVE_RUNNER)
        if(NOT SEGACD_BIOS_PROBE_Z80_MODULE_DIR)
          message(FATAL_ERROR "The native runner requires a generated Z80 module")
        endif()
        enable_language(CXX)
        if(WIN32)
          if(NOT SDL2_DIR)
            set(SDL2_DIR "${genesis_root}/runner/external/SDL2/SDL2-2.28.5/cmake")
          endif()
          find_package(SDL2 REQUIRED CONFIG NO_DEFAULT_PATH PATHS "${SDL2_DIR}")
        else()
          find_package(SDL2 REQUIRED)
        endif()
        get_target_property(native_catalog_sources segacd_bios_dual_probe SOURCES)
        list(FILTER native_catalog_sources EXCLUDE REGEX "(bios_dual_probe|bios_services|bios_tables|segacd_coordinator|genesis_bus|genesis_vdp|z80_native|m68k_step)\\.c$")
        get_target_property(native_includes segacd_bios_dual_probe INCLUDE_DIRECTORIES)
        get_target_property(native_definitions segacd_bios_dual_probe COMPILE_DEFINITIONS)
        add_library(segacd_owned_native_program STATIC ${native_catalog_sources})
        target_include_directories(segacd_owned_native_program PRIVATE ${native_includes})
        target_compile_features(segacd_owned_native_program PRIVATE c_std_11)
        if(NOT SEGACD_GAME_TARGET OR NOT SEGACD_GAME_SOURCES)
          message(FATAL_ERROR "A native runner requires SEGACD_GAME_TARGET and SEGACD_GAME_SOURCES from its consumer.")
        endif()
        add_executable(${SEGACD_GAME_TARGET} ${SEGACD_GAME_SOURCES}
          "${cd_runner}/bios_services.c" "${cd_runner}/bios_tables.c"
          "${cd_runner}/backup.c" "${cd_runner}/audio_mix.c" "${cd_runner}/audio_native.c" "${cd_runner}/debug.c"
          "${genesis_root}/runner/video/genesis_raster_jobs.c" "${genesis_root}/runner/video/genesis_hd_gpu.c"
          "${genesis_root}/runner/audio.c" "${genesis_root}/runner/engine_worker.c" "${genesis_root}/runner/host_stream.c" "${genesis_root}/runner/persist_writer.c" "${genesis_root}/runner/host_atomic_file.c"
          "${genesis_root}/runner/audio/sn76489.c" "${genesis_root}/runner/audio/ym2612_ymfm.cpp"
          "${genesis_root}/runner/external/ymfm/src/ymfm_opn.cpp" "${genesis_root}/runner/external/ymfm/src/ymfm_ssg.cpp"
          "${genesis_root}/runner/external/ymfm/src/ymfm_adpcm.cpp" "${genesis_root}/runner/external/ymfm/src/ymfm_pcm.cpp")
        include("${cd_runner}/recomp_ui.cmake")
        segacd_target_recomp_ui(${SEGACD_GAME_TARGET})
        target_include_directories(${SEGACD_GAME_TARGET} PRIVATE ${native_includes} "${genesis_root}/runner" "${genesis_root}/runner/include"
          "${genesis_root}/runner/external/ymfm/src" "${genesis_root}/runner/video" "${cd_runner}")
        foreach(sdl_include IN LISTS SDL2_INCLUDE_DIRS)
          get_filename_component(sdl_name "${sdl_include}" NAME)
          if(sdl_name STREQUAL "SDL2")
            get_filename_component(sdl_parent "${sdl_include}" DIRECTORY)
            target_include_directories(${SEGACD_GAME_TARGET} PRIVATE "${sdl_parent}")
          endif()
        endforeach()
        option(GENESIS_ENGINE_WORKERS "Build optional persistent host workers" ON)
        target_compile_definitions(${SEGACD_GAME_TARGET} PRIVATE ${native_definitions} _CRT_SECURE_NO_WARNINGS
          SEGACD_IMMUTABLE_CODE=$<BOOL:${SEGACD_IMMUTABLE_CODE}>
          GENESIS_HD_JOBS=1
          GENESIS_ENGINE_WORKERS=$<BOOL:${GENESIS_ENGINE_WORKERS}>)
        target_compile_definitions(segacd_machine PRIVATE _CRT_SECURE_NO_WARNINGS GENESIS_HD_JOBS=1
          GENESIS_ENGINE_WORKERS=$<BOOL:${GENESIS_ENGINE_WORKERS}>)
        target_compile_features(${SEGACD_GAME_TARGET} PRIVATE c_std_11 cxx_std_17)
        set(SEGACD_COMPACT_CATALOG_DIR "" CACHE PATH "Qualified generic compact catalog metadata; empty retains the original registry")
        if(SEGACD_COMPACT_CATALOG_DIR)
          if(NOT EXISTS "${SEGACD_COMPACT_CATALOG_DIR}/native_catalog_manifest.json" OR
             NOT EXISTS "${SEGACD_COMPACT_CATALOG_DIR}/native_catalog_registry.h")
            message(FATAL_ERROR "Compact catalog bundle is incomplete; regenerate with tools/native_catalog_compact.py")
          endif()
          include("${SEGACD_COMPACT_CATALOG_DIR}/native_catalog_sources.cmake")
          target_sources(${SEGACD_GAME_TARGET} PRIVATE ${GENESIS_COMPACT_CATALOG_SOURCES})
          target_include_directories(${SEGACD_GAME_TARGET} PRIVATE "${SEGACD_COMPACT_CATALOG_DIR}")
          target_compile_definitions(${SEGACD_GAME_TARGET} PRIVATE GENESIS_COMPACT_NATIVE_CATALOGS=1)
        endif()
        target_link_libraries(${SEGACD_GAME_TARGET} PRIVATE segacd_owned_native_program segacd_machine segacd_disc_image SDL2::SDL2)
        target_link_libraries(${SEGACD_GAME_TARGET} PRIVATE recomp-audio::core recomp-persistence-core::core sega-psg::core retro-runtime::presentation recomp-input::core recomp-diagnostics-core::core)
        include("${genesis_root}/cmake/GenesisRecompOptimization.cmake")
        foreach(native_target ${SEGACD_GAME_TARGET} segacd_owned_native_program segacd_machine segacd_devices segacd_disc_image)
          genesisrecomp_native_optimization(${native_target})
        endforeach()
        if(TARGET SDL2::SDL2main)
          target_link_libraries(${SEGACD_GAME_TARGET} PRIVATE SDL2::SDL2main)
        endif()
        if(NOT WIN32)
          target_link_libraries(${SEGACD_GAME_TARGET} PRIVATE m)
        else()
          target_link_libraries(${SEGACD_GAME_TARGET} PRIVATE ws2_32)
          set_target_properties(${SEGACD_GAME_TARGET} PROPERTIES WIN32_EXECUTABLE TRUE)
          add_custom_command(TARGET ${SEGACD_GAME_TARGET} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "$<TARGET_FILE:SDL2::SDL2>" "$<TARGET_FILE_DIR:${SEGACD_GAME_TARGET}>" VERBATIM)
        endif()
      endif()
    endif()
  endif()
endif()
